// ================================= // SEARCH // ================================= if (action === "search") { var searchText = String(e.parameter.search || "") .trim() .toLowerCase(); var searchType = String(e.parameter.searchType || "all") .trim(); if (searchText === "") { return sendJSON({ status: "error", message: "Please enter search value" }); } var data = sheet.getDataRange().getValues(); var records = []; // ================================= // COLUMN INDEX // ================================= var fieldIndex = { fullName: 0, fatherName: 1, mobile: 2, email: 3, address: 4, userId: 5 }; // ================================= // SEARCH DATA // ================================= for (var i = 1; i < data.length; i++) { var match = false; // --------------------------------- // ALL FIELDS // --------------------------------- if (searchType === "all") { for (var key in fieldIndex) { var value = String( data[i][fieldIndex[key]] || "" ) .trim() .toLowerCase(); if (value.indexOf(searchText) !== -1) { match = true; break; } } } // --------------------------------- // SELECTED FIELD // --------------------------------- else { if (fieldIndex[searchType] === undefined) { return sendJSON({ status: "error", message: "Invalid Search Field" }); } var value = String( data[i][fieldIndex[searchType]] || "" ) .trim() .toLowerCase(); if (value.indexOf(searchText) !== -1) { match = true; } } // ================================= // ADD RECORD // ================================= if (match) { records.push({ rowNumber: i + 1, fullName: String(data[i][0] || ""), fatherName: String(data[i][1] || ""), mobile: String(data[i][2] || ""), email: String(data[i][3] || ""), address: String(data[i][4] || ""), userId: String(data[i][5] || "") }); } } // ================================= // NO RECORD // ================================= if (records.length === 0) { return sendJSON({ status: "error", message: "No Record Found" }); } // ================================= // SUCCESS // ================================= return sendJSON({ status: "success", records: records }); }
